function doPost(e) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); const data = JSON.parse(e.postData.contents); const values = sheet.getDataRange().getValues(); // проверка дубля по телефону for (let i = 1; i < values.length; i++) { if (values[i][3] === data.phone) { return ContentService.createTextOutput(JSON.stringify({ status: "duplicate" })); } } const currentCount = values.length; // включая заголовок const nextNumber = currentCount; // 1..100 if (nextNumber > 100) { return ContentService.createTextOutput(JSON.stringify({ status: "full" })); } const formattedNumber = String(nextNumber).padStart(3, "0"); sheet.appendRow([ new Date(), data.name, data.surname, data.phone, data.email, data.club, formattedNumber ]); return ContentService.createTextOutput(JSON.stringify({ status: "ok", number: formattedNumber })); }
