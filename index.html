function doPost(e) { try { const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); const data = JSON.parse(e.postData.contents); const phone = data.phone; // Проверка на дубликат телефона const lastRow = sheet.getLastRow(); if (lastRow >= 2) { const phones = sheet.getRange(2, 4, lastRow - 1, 1).getValues().flat(); if (phones.includes(phone)) { // Возвращаем ошибку, но из-за no-cors мы её не увидим, запись не произойдёт return ContentService .createTextOutput(JSON.stringify({ success: false, error: "duplicate" })) .setMimeType(ContentService.MimeType.JSON); } } // Находим следующий ID let newId = 1; if (lastRow >= 2) { const ids = sheet.getRange(2, 1, lastRow - 1, 1).getValues().flat().filter(id => !isNaN(id) && id > 0); newId = ids.length > 0 ? Math.max(...ids) + 1 : 1; } sheet.appendRow([newId, data.name, data.email || '', phone, data.timestamp, data.interest || '']); return ContentService .createTextOutput(JSON.stringify({ success: true, id: newId })) .setMimeType(ContentService.MimeType.JSON); } catch(error) { return ContentService .createTextOutput(JSON.stringify({ success: false, error: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
